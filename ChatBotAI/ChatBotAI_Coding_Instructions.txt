```python
import requests
import json

OLLAMA_URL = "http://localhost:11434/api/generate"
MODEL_NAME = "mistral:7b-instruct"

def generate_response(prompt: str) -> str:
    """Send prompt to Ollama and return response text."""
    payload = {
        "model": MODEL_NAME,
        "prompt": prompt,
        "stream": False
    }

    response = requests.post(OLLAMA_URL, json=payload)
    response.raise_for_status()

    return response.json().get("response", "")

def extract_guest_info(message_content: str) -> dict:
    """Extract structured guest information from a message."""

    extraction_prompt = f'''Analyze this message and extract guest information.

MESSAGE:
{message_content}

Extract ONLY what is explicitly mentioned. Return valid JSON:
{{
    "guest_name": "string or null",
    "family_members": [{{"name": "string", "relation": "string"}}],
    "pets": [{{"name": "string", "type": "string"}}],
    "preferences": [{{"type": "string", "value": "string"}}],
    "check_in": "date or null",
    "check_out": "date or null",
    "num_guests": "number or null",
    "special_requests": ["string"],
    "mentioned_interests": ["string"]
}}

Return ONLY the JSON, no explanation.'''

    response = generate_response(extraction_prompt)

    try:
        # Parse JSON from response
        return json.loads(response)
    except json.JSONDecodeError:
        # If parsing fails, return empty structure
        return {}

def generate_guest_response(guest_profile: dict, conversation_history: list,
                            new_message: str, property_info: dict) -> str:
    """Generate a response using guest context."""

    prompt = f'''You are a helpful assistant for a vacation rental business.

GUEST PROFILE:
- Name: {guest_profile.get('name', 'Guest')}
- Previous stays: {guest_profile.get('total_stays', 0)}
- Family members: {guest_profile.get('family', 'Unknown')}
- Pets: {guest_profile.get('pets', 'None mentioned')}
- Preferences: {guest_profile.get('preferences', 'None recorded')}
- Past notes: {guest_profile.get('notes', 'None')}

PROPERTY INFORMATION:
{json.dumps(property_info, indent=2)}

CONVERSATION HISTORY:
{format_conversation(conversation_history)}

NEW MESSAGE FROM GUEST:
{new_message}

INSTRUCTIONS:
- Be warm, professional, and helpful
- If returning guest, welcome them back naturally
- Reference relevant details from their profile (family, pets) naturally
- Answer questions accurately based on property info
- If unsure, offer to check and confirm
- Keep response concise but friendly

YOUR RESPONSE:'''

    return generate_response(prompt)

def format_conversation(messages: list) -> str:
    """Format message history for prompt."""
    formatted = []
    for msg in messages[-10:]:  # Last 10 messages for context
        sender = msg.get('sender_type', 'unknown').upper()
        content = msg.get('content', '')
        formatted.append(f"{sender}: {content}")
    return "\n".join(formatted)
```

================================================================================
                         MEMORY SERVICE IMPLEMENTATION
================================================================================

File: services/memory_service.py

This service handles the CRITICAL guest memory feature.

```python
from models import db, Guest, GuestDetail, Message
from services.ai_service import extract_guest_info

def process_message_for_memory(message: Message):
    """
    Process a message to extract and store guest information.
    This runs for EVERY message, regardless of who sent it.
    """

    if message.is_processed:
        return

    # Extract information using AI
    extracted = extract_guest_info(message.content)

    if not extracted:
        message.is_processed = True
        db.session.commit()
        return

    guest = message.conversation.guest

    # Store family members
    for family in extracted.get('family_members', []):
        if family.get('name'):
            store_detail(
                guest_id=guest.id,
                detail_type='family',
                detail_key=family.get('relation', 'member'),
                detail_value=family.get('name'),
                source_message_id=message.id
            )

    # Store pets
    for pet in extracted.get('pets', []):
        if pet.get('name'):
            store_detail(
                guest_id=guest.id,
                detail_type='pet',
                detail_key=pet.get('type', 'pet'),
                detail_value=pet.get('name'),
                source_message_id=message.id
            )

    # Store preferences
    for pref in extracted.get('preferences', []):
        store_detail(
            guest_id=guest.id,
            detail_type='preference',
            detail_key=pref.get('type', 'general'),
            detail_value=pref.get('value'),
            source_message_id=message.id
        )

    # Store special requests
    for request in extracted.get('special_requests', []):
        store_detail(
            guest_id=guest.id,
            detail_type='special_request',
            detail_key='request',
            detail_value=request,
            source_message_id=message.id
        )

    # Mark message as processed
    message.is_processed = True
    db.session.commit()

def store_detail(guest_id: int, detail_type: str, detail_key: str,
                 detail_value: str, source_message_id: int = None):
    """Store a guest detail, avoiding duplicates."""

    # Check if this exact detail already exists
    existing = GuestDetail.query.filter_by(
        guest_id=guest_id,
        detail_type=detail_type,
        detail_key=detail_key,
        detail_value=detail_value
    ).first()

    if existing:
        return  # Already stored

    detail = GuestDetail(
        guest_id=guest_id,
        detail_type=detail_type,
        detail_key=detail_key,
        detail_value=detail_value,
        source_message_id=source_message_id
    )
    db.session.add(detail)
    db.session.commit()

def get_guest_profile(guest_id: int) -> dict:
    """Build a complete guest profile from stored details."""

    guest = Guest.query.get(guest_id)
    if not guest:
        return {}

    details = GuestDetail.query.filter_by(guest_id=guest_id).all()

    profile = {
        'id': guest.id,
        'name': guest.name,
        'email': guest.email,
        'phone': guest.phone,
        'total_stays': guest.total_stays,
        'first_contact': str(guest.first_contact),
        'last_contact': str(guest.last_contact),
        'notes': guest.notes,
        'family': [],
        'pets': [],
        'preferences': [],
        'special_requests': []
    }

    for detail in details:
        if detail.detail_type == 'family':
            profile['family'].append({
                'relation': detail.detail_key,
                'name': detail.detail_value
            })
        elif detail.detail_type == 'pet':
            profile['pets'].append({
                'type': detail.detail_key,
                'name': detail.detail_value
            })
        elif detail.detail_type == 'preference':
            profile['preferences'].append({
                'type': detail.detail_key,
                'value': detail.detail_value
            })
        elif detail.detail_type == 'special_request':
            profile['special_requests'].append(detail.detail_value)

    return profile
```

================================================================================
                         MESSAGE FLOW - HOW IT WORKS
================================================================================

INCOMING MESSAGE FLOW:

1. Message arrives (email, WhatsApp, etc.)
       |
       v
2. message_router.py receives it
       |
       v
3. Identify or create guest (by email, phone, platform ID)
       |
       v
4. Create/update Conversation record
       |
       v
5. Store Message in database
       |
       v
6. memory_service.process_message_for_memory()
   - Extract info using AI
   - Store in GuestDetail table
       |
       v
7. Check if AI auto-response is enabled
       |
       +-- YES --> ai_service.generate_guest_response()
       |           - Load guest profile
       |           - Include in prompt
       |           - Generate personalized response
       |           - Send response via platform
       |
       +-- NO --> Just notify owner in dashboard

OWNER SENDS MESSAGE FLOW:

1. Owner writes message in dashboard
       |
       v
2. Store Message (sender_type='owner')
       |
       v
3. Send via appropriate platform API
       |
       v
4. memory_service.process_message_for_memory()
   - Extract info from OWNER'S message too
   - Sometimes owner mentions guest details
       |
       v
5. Done
